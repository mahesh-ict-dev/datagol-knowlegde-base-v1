# Examples

This section provides comprehensive examples of how to use the Orchestration API for common use cases.

## Example 1: ETL Pipeline Workflow

### Step 1: Create DAG

```bash
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "ETL Pipeline",
    "scheduleType": "SCHEDULED",
    "schedule": {
      "basicSchedule": {
        "timeUnit": "HOUR",
        "units": "24"
      }
    },
    "defaultParameters": {
      "sourceDatabase": "production",
      "targetDatabase": "warehouse"
    },
    "metadata": {
      "description": "Daily ETL pipeline for data warehouse",
      "owner": "data-team",
      "environment": "production"
    }
  }'
```

**Response:**
```json
{
  "id": 1,
  "name": "ETL Pipeline",
  "companyId": 123,
  "createdAt": "2024-01-15T10:30:00Z",
  "createdBy": {
    "id": 456,
    "email": "user@example.com"
  },
  "scheduleType": "SCHEDULED",
  "schedule": {
    "basicSchedule": {
      "timeUnit": "HOUR",
      "units": "24"
    }
  }
}
```

### Step 2: Add Nodes

```bash
# Add extraction node
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags/1/nodes" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "pipelineId": 101,
    "metadata": {
      "step": "extract",
      "description": "Extract data from source database",
      "priority": "high",
      "timeout": 3600,
      "retryCount": 3
    }
  }'

# Add transformation node
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags/1/nodes" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "pipelineId": 102,
    "metadata": {
      "step": "transform",
      "description": "Transform and clean data",
      "priority": "medium",
      "timeout": 7200,
      "retryCount": 2
    }
  }'

# Add load node
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags/1/nodes" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "pipelineId": 103,
    "metadata": {
      "step": "load",
      "description": "Load data into warehouse",
      "priority": "high",
      "timeout": 1800,
      "retryCount": 3
    }
  }'
```

### Step 3: Create Edges

```bash
# Connect extract to transform
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags/1/edges" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "fromNodeId": 1,
    "toNodeId": 2,
    "metadata": {
      "condition": "success",
      "description": "Data flow from extraction to transformation",
      "priority": 1
    }
  }'

# Connect transform to load
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags/1/edges" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "fromNodeId": 2,
    "toNodeId": 3,
    "metadata": {
      "condition": "success",
      "description": "Data flow from transformation to load",
      "priority": 1
    }
  }'
```

### Step 4: Execute DAG

```bash
curl -X POST "https://api.saasxl.com/orchestration/api/v1/dags/1/runs" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "parameters": {
      "sourceTable": "users",
      "targetTable": "dim_users",
      "batchSize": 1000,
      "extractDate": "2024-01-15"
    }
  }'
```

## Example 8: Python Client Implementation

```python
import requests
import json
from typing import Dict, List, Optional

class OrchestrationClient:
    def __init__(self, base_url: str, token: str):
        self.base_url = base_url
        self.headers = {
            'Authorization': f'Bearer {token}',
            'Content-Type': 'application/json'
        }
    
    def create_dag(self, name: str, schedule_type: str, schedule_config: Dict, 
                   default_params: Optional[Dict] = None, metadata: Optional[Dict] = None) -> Dict:
        """Create a new DAG workflow."""
        url = f"{self.base_url}/orchestration/api/v1/dags"
        data = {
            "name": name,
            "scheduleType": schedule_type,
            "schedule": schedule_config
        }
        
        if default_params:
            data["defaultParameters"] = default_params
        if metadata:
            data["metadata"] = metadata
            
        response = requests.post(url, headers=self.headers, json=data)
        response.raise_for_status()
        return response.json()
    
    def add_node(self, dag_id: int, pipeline_id: int, metadata: Optional[Dict] = None) -> Dict:
        """Add a node to a DAG."""
        url = f"{self.base_url}/orchestration/api/v1/dags/{dag_id}/nodes"
        data = {
            "pipelineId": pipeline_id,
            "metadata": metadata or {}
        }
        response = requests.post(url, headers=self.headers, json=data)
        response.raise_for_status()
        return response.json()
    
    def create_edge(self, dag_id: int, from_node_id: int, to_node_id: int, 
                   metadata: Optional[Dict] = None) -> Dict:
        """Create an edge between two nodes."""
        url = f"{self.base_url}/orchestration/api/v1/dags/{dag_id}/edges"
        data = {
            "fromNodeId": from_node_id,
            "toNodeId": to_node_id,
            "metadata": metadata or {}
        }
        response = requests.post(url, headers=self.headers, json=data)
        response.raise_for_status()
        return response.json()
    
    def run_dag(self, dag_id: int, parameters: Optional[Dict] = None) -> Dict:
        """Execute a DAG run."""
        url = f"{self.base_url}/orchestration/api/v1/dags/{dag_id}/runs"
        data = {"parameters": parameters or {}}
```

```javascript
class OrchestrationClient {
    constructor(baseUrl, token) {
        this.baseUrl = baseUrl;
        this.headers = {
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json'
        };
    }

    async createDag(name, scheduleType, scheduleConfig, defaultParams = {}, metadata = {}) {
        const url = `${this.baseUrl}/orchestration/api/v1/dags`;
        const data = {
            name,
            scheduleType,
            schedule: scheduleConfig,
            defaultParameters: defaultParams,
            metadata
        };
        
        const response = await fetch(url, {
            method: 'POST',
            headers: this.headers,
            body: JSON.stringify(data)
        });
        
        if (!response.ok) {
            throw new Error(`Failed to create DAG: ${response.statusText}`);
        }
        
        return response.json();
    }

    async addNode(dagId, pipelineId, metadata = {}) {
        const url = `${this.baseUrl}/orchestration/api/v1/dags/${dagId}/nodes`;
        const data = {
            pipelineId,
            metadata
        };
        
        const response = await fetch(url, {
            method: 'POST',
            headers: this.headers,
            body: JSON.stringify(data)
        });
        
        if (!response.ok) {
            throw new Error(`Failed to add node: ${response.statusText}`);
        }
        
        return response.json();
    }

    async createEdge(dagId, fromNodeId, toNodeId, metadata = {}) {
        const url = `${this.baseUrl}/orchestration/api/v1/dags/${dagId}/edges`;
        const data = {
            fromNodeId,
            toNodeId,
            metadata
        };
        
        const response = await fetch(url, {
            method: 'POST',
            headers: this.headers,
            body: JSON.stringify(data)
        });
        
        if (!response.ok) {
            throw new Error(`Failed to create edge: ${response.statusText}`);
        }
        
        return response.json();
    }

    async runDag(dagId, parameters = {}) {
        const url = `${this.baseUrl}/orchestration/api/v1/dags/${dagId}/runs`;
        const data = { parameters };
        
        const response = await fetch(url, {
            method: 'POST',
            headers: this.headers,
            body: JSON.stringify(data)
        });
        
        if (!response.ok) {
            throw new Error(`Failed to run DAG: ${response.statusText}`);
        }
        
        return response.json();
    }

    async getDagStatus(dagId) {
        const url = `${this.baseUrl}/orchestration/api/v1/dags/${dagId}`;
        const response = await fetch(url, {
            method: 'GET',
            headers: this.headers
        });
        
        if (!response.ok) {
            throw new Error(`Failed to get DAG status: ${response.statusText}`);
        }
        
        return response.json();
    }

    async getDagRuns(dagId, includeJobs = false, limit = 50) {
        const url = `${this.baseUrl}/orchestration/api/v1/dags/${dagId}/runs`;
        const params = new URLSearchParams({
            includeJobs: includeJobs.toString(),
            limit: limit.toString()
        });
        
        const response = await fetch(`${url}?${params}`, {
            method: 'GET',
            headers: this.headers
        });
        
        if (!response.ok) {
            throw new Error(`Failed to get DAG runs: ${response.statusText}`);
        }
        
        return response.json();
    }
```
