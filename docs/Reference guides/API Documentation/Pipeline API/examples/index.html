Examples

This section provides practical examples of how to use the Pipeline API.

Creating a Pipeline​

Basic Pipeline Creation​

curl -X POST "https://api.example.com/dataPipeline/api/v2/" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Daily Sales Report",
    "type": "STANDARD",
    "description": "Generates daily sales reports",
    "scheduleType": "CRON",
    "schedule": {
      "cronExpression": "0 2 * * *",
      "timeZone": "UTC"
    },
    "sources": [
      {
        "id": 123,
        "name": "Sales Database",
        "type": "POSTGRESQL"
      }
    ],
    "destination": {
      "id": 456,
      "name": "Data Warehouse",
      "type": "SNOWFLAKE"
    },
    "streams": [
      {
        "name": "sales_summary",
        "query": "SELECT date, SUM(amount) FROM sales GROUP BY date",
        "outputTableName": "daily_sales_summary",
        "syncMode": "FULL"
      }
    ],
    "queryEngine": "SPARK"
  }'

JavaScript Examples​

Creating a Pipeline with JavaScript​

async function createPipeline(pipelineData) {
  const response = await fetch('/dataPipeline/api/v2/', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(pipelineData)
  });

  if (!response.ok) {
    throw new Error(`Failed to create pipeline: ${response.statusText}`);
  }

  return await response.json();
}

// Usage
const pipelineData = {
  name: 'E-commerce Analytics',
  type: 'STANDARD',
  description: 'Processes e-commerce transaction data',
  scheduleType: 'CRON',
  schedule: {
    cronExpression: '0 */4 * * *', // Every 4 hours
    timeZone: 'UTC'
  },
  sources: [
    {
      id: 234,
      name: 'E-commerce Database',
      type: 'POSTGRESQL'
    }
  ],
  destination: {
    id: 567,
    name: 'Analytics Platform',
    type: 'SNOWFLAKE'
  },
  streams: [
    {
      name: 'transactions',
      query: 'SELECT * FROM transactions WHERE created_at >= :start_date',
      outputTableName: 'processed_transactions',
      syncMode: 'INCREMENTAL',
      columns: ['id', 'user_id', 'amount', 'product_id', 'created_at'],
      uniqueColumns: ['id'],
      requiredParams: ['start_date']
    }
  ],
  parameters: {
    start_date: '2024-01-01T00:00:00Z'
  },
  queryEngine: 'SPARK'
};

createPipeline(pipelineData)
  .then(pipeline => console.log('Pipeline created:', pipeline))
  .catch(error => console.error('Error:', error));

Creating a Pipeline with Python​

import requests
import json
from datetime import datetime

def create_pipeline(pipeline_data, token):
    headers = {

Running a Pipeline with Python​

import requests
import time

def run_pipeline(pipeline_id, parameters=None, token=None):
    headers = {
        'Authorization': f'Bearer {token}',
        'Content-Type': 'application/json'
    }
    
    data = parameters or {}
    
    response = requests.post(
        f'/dataPipeline/api/v1/{pipeline_id}/run',
        json=data,
        headers=headers
    )
    
    response.raise_for_status()
    return response.json()

def wait_for_job_completion(pipeline_id, job_id, token, timeout=3600):
    """Wait for job completion with timeout"""
    start_time = time.time()
    
    while time.time() - start_time < timeout:
        response = requests.get(
            f'/dataPipeline/api/v1/{pipeline_id}/job/{job_id}',
            headers={'Authorization': f'Bearer {token}'}
        )
        
        if response.ok:
            job = response.json()
            status = job['status']
            
            if status in ['COMPLETED', 'SUCCEEDED']:
                return job
            elif status in ['FAILED', 'CANCELLED']:
                raise Exception(f"Job failed with status: {status}")
        
        time.sleep(10)  # Wait 10 seconds before checking again
    
    raise Exception("Job timeout")

# Usage
try:
    # Run pipeline
    job = run_pipeline(123, {
        'date': datetime.now().strftime('%Y-%m-%d'),
        'region': 'US'
    }, 'your-jwt-token')
    
    print(f"Job started: {job['id']}")
    
    # Wait for completion
    completed_job = wait_for_job_completion(123, job['id'], 'your-jwt-token')
    print(f"Job completed: {completed_job['status']}")
    
except Exception as e:
    print(f"Error: {e}")

Monitoring Pipeline Jobs with Python​

import requests
from datetime import datetime, timedelta

def get_pipeline_jobs(status=None, time_range=None, token=None):
    headers = {
        'Authorization': f'Bearer {token}',
        'Content-Type': 'application/json'
    }
    
    request_body = {}
    
    if status:
        request_body['status'] = status
    
    if time_range:
        request_body['time'] = time_range
    
    response = requests.post(
        '/dataPipeline/api/v2/job',
        json=request_body,
        headers=headers
    )
    
    response.raise_for_status()
    return response.json()

def get_failed_jobs(hours=24, token=None):
    """Get failed jobs from the last N hours"""
    response = requests.get(
        f'/dataPipeline/api/v1/getAllFailedJobs/{hours}/HOUR',
        headers={'Authorization': f'Bearer {token}'}
    )
    
    response.raise_for_status()
    return response.json()

def get_job_error_log(pipeline_id, job_id, token=None):
    """Get error logs for a specific job"""
    response = requests.get(
        f'/dataPipeline/api/v1/{pipeline_id}/job/{job_id}/errorLog',
        headers={'Authorization': f'Bearer {token}'}
    )
    
    response.raise_for_status()
    return response.json()

# Usage
token = 'your-jwt-token'

# Get running jobs
running_jobs = get_pipeline_jobs(status='RUNNING', token=token)
print(f"Running jobs: {len(running_jobs['jobs'])}")

# Get failed jobs from last 24 hours
failed_jobs = get_failed_jobs(hours=24, token=token)
print(f"Failed jobs in last 24 hours: {len(failed_jobs['jobs'])}")

# Get error logs for a specific job
if failed_jobs['jobs']:
    first_failed_job = failed_jobs['jobs'][0]
    pipeline_id = first_failed_job['pipelineInfo']['pipelineId']
    job_id = first_failed_job['job']['id']
    
    error_logs = get_job_error_log(pipeline_id, job_id, token)
    print(f"Error logs for job {job_id}:")
    for log in error_logs['errorLogs']:
        print(f"  {log['timestamp']}: {log['message']}")

React Hook Example​

import { useState, useEffect } from 'react';

interface Pipeline {
  id: number;
  name: string;
  type: string;
  description: string;
  scheduleType: string;
  status: string;
}

interface UsePipelinesReturn {
  pipelines: Pipeline[];
  loading: boolean;
  error: string | null;
  createPipeline: (pipelineData: Partial<Pipeline>) => Promise<void>;
  updatePipeline: (id: number, pipelineData: Partial<Pipeline>) => Promise<void>;
  deletePipeline: (id: number) => Promise<void>;
  runPipeline: (id: number, parameters?: any) => Promise<void>;
}

export function usePipelines(token: string): UsePipelinesReturn {
  const [pipelines, setPipelines] = useState<Pipeline[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchPipelines = async () => {
    try {
      const response = await fetch('/dataPipeline/api/v2/', {
        headers: {
          'Authorization': `Bearer ${token}`
        }
      });

      if (!response.ok) {
        throw new Error('Failed to fetch pipelines');
      }

      const data = await response.json();
      setPipelines(data);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    } finally {
      setLoading(false);
    }
  };

  const createPipeline = async (pipelineData: Partial<Pipeline>) => {
    try {
      const response = await fetch('/dataPipeline/api/v2/', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(pipelineData)
      });

      if (!response.ok) {
        throw new Error('Failed to create pipeline');
      }

      await fetchPipelines(); // Refresh the list
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    }
  };

  const updatePipeline = async (id: number, pipelineData: Partial<Pipeline>) => {
    try {
      const response = await fetch(`/dataPipeline/api/v2/${id}`, {
        method: 'PUT',
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(pipelineData)
      });

      if (!response.ok) {
        throw new Error('Failed to update pipeline');
      }

      await fetchPipelines(); // Refresh the list
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    }
  };

  const deletePipeline = async (id: number) => {
    try {
      const response = await fetch(`/dataPipeline/api/v1/${id}`, {
        method: 'DELETE',
        headers: {
          'Authorization': `Bearer ${token}`
        }
      });

      if (!response.ok) {
        throw new Error('Failed to delete pipeline');
      }

      await fetchPipelines(); // Refresh the list
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    }
  };

  const runPipeline = async (id: number, parameters: any = {}) => {
    try {
      const response = await fetch(`/dataPipeline/api/v1/${id}/run`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(parameters)
      });

      if (!response.ok) {
        throw new Error('Failed to run pipeline');
      }

      const job = await response.json();
      console.log('Pipeline job started:', job);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    }
  };

  useEffect(() => {
    fetchPipelines();
  }, [token]);

  return {
    pipelines,
    loading,
    error,
    createPipeline,
    updatePipeline,
    deletePipeline,
    runPipeline
  };
}

// Usage in React component
function PipelineList() {
  const { pipelines, loading, error, createPipeline, runPipeline } = usePipelines('your-jwt-token');

  if (loading) return <div>Loading pipelines...</div>;
  if (error) return <div>Error: {error}</div>;

  return (
    <div>
      <h2>Pipelines</h2>
      {pipelines.map(pipeline => (
        <div key={pipeline.id}>
          <h3>{pipeline.name}</h3>
          <p>{pipeline.description}</p>
          <button onClick={() => runPipeline(pipeline.id)}>
            Run Pipeline
          </button>
        </div>
      ))}
    </div>
  );
}

Real-time Pipeline Monitoring​

// Real-time pipeline monitoring with WebSocket
class PipelineMonitor {
  constructor(token) {
    this.token = token;
    this.ws = null;
    this.callbacks = new Map();
  }

  connect() {
    this.ws = new WebSocket(`wss://api.example.com/pipeline-monitor?token=${this.token}`);
    
    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      this.handleMessage(data);
    };

    this.ws.onclose = () => {
      console.log('Pipeline monitor disconnected');
      // Reconnect after 5 seconds
      setTimeout(() => this.connect(), 5000);
    };
  }

  handleMessage(data) {
    switch (data.type) {
      case 'JOB_STARTED':
        this.notifyCallbacks('jobStarted', data);
        break;
      case 'JOB_COMPLETED':
        this.notifyCallbacks('jobCompleted', data);
        break;
      case 'JOB_FAILED':
        this.notifyCallbacks('jobFailed', data);
        break;
      case 'PIPELINE_STATUS_CHANGED':
        this.notifyCallbacks('pipelineStatusChanged', data);
        break;
    }
  }

  on(event, callback) {
    if (!this.callbacks.has(event)) {
      this.callbacks.set(event, []);
    }
    this.callbacks.get(event).push(callback);
  }

  notifyCallbacks(event, data) {
    const callbacks = this.callbacks.get(event) || [];
    callbacks.forEach(callback => callback(data));
  }

  disconnect() {
    if (this.ws) {
      this.ws.close();
    }
  }
}

// Usage
const monitor = new PipelineMonitor('your-jwt-token');

monitor.on('jobStarted', (data) => {
  console.log(`Job ${data.jobId} started for pipeline ${data.pipelineId}`);
});

monitor.on('jobCompleted', (data) => {
  console.log(`Job ${data.jobId} completed successfully`);
});

monitor.on('jobFailed', (data) => {
  console.error(`Job ${data.jobId} failed: ${data.error}`);
  // Send notification or alert
});

monitor.connect();
