API Examples

This section provides comprehensive examples for using the NoCoDB API across all versions.

NoCoDB v1 Examples​

Creating a Transformation​

curl -X POST "noCo/api/v1/transformation" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Sales Data Transformation",
    "description": "Transform sales data from CSV to structured format",
    "sourceTable": "raw_sales",
    "targetTable": "processed_sales",
    "transformationRules": [
      {
        "sourceColumn": "date",
        "targetColumn": "sale_date",
        "transformation": "DATE_PARSE"
      },
      {
        "sourceColumn": "amount",
        "targetColumn": "sale_amount",
        "transformation": "TO_DECIMAL"
      }
    ]
  }'

Creating a Widget​

curl -X POST "noCo/api/v1/widgets" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Monthly Sales Chart",
    "type": "CHART",
    "config": {
      "chartType": "LINE",
      "dataSource": "sales_table",
      "xAxis": "month",
      "yAxis": "total_sales",
      "title": "Monthly Sales Trend"
    }
  }'

Uploading a CSV File​

curl -X POST "noCo/api/v1/file/upload/csv" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -F "file=@sales_data.csv" \
  -F "tableName=sales_data" \
  -F "description=Monthly sales data for 2024"

Creating a Webhook​

curl -X POST "noCo/api/v1/tables/webhooks" \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Sales Alert Webhook",
    "url": "https://api.example.com/webhooks/sales",
    "events": ["CREATE", "UPDATE"],
    "tableId": "sales_table",
    "headers": {
      "Authorization": "Bearer webhook_token"
    },
    "active": true
  }'

NoCoDB v2 Client​

class NoCoDBClient {
  constructor(baseUrl, token) {
    this.baseUrl = baseUrl;
    this.token = token;
    this.headers = {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    };
  }

  async createWorkspace(workspaceData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(workspaceData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create workspace: ${response.statusText}`);
    }

    return await response.json();
  }

  async createTable(workspaceId, tableData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(tableData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create table: ${response.statusText}`);
    }

    return await response.json();
  }

  async addRows(workspaceId, tableId, rows) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables/${tableId}/rows/bulk`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(rows)
    });

    if (!response.ok) {
      throw new Error(`Failed to add rows: ${response.statusText}`);
    }

    return await response.json();
  }

  async getTables(workspaceId, page = 0, size = 20) {
    const response = await fetch(
      `${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables?page=${page}&size=${size}`,
      {
        headers: this.headers
      }
    );

    if (!response.ok) {
      throw new Error(`Failed to get tables: ${response.statusText}`);
    }

    return await response.json();
  }

  async updateTable(workspaceId, tableId, tableData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables/${tableId}`, {
      method: 'PUT',
      headers: this.headers,
      body: JSON.stringify(tableData)
    });

    if (!response.ok) {
      throw new Error(`Failed to update table: ${response.statusText}`);
    }

    return await response.json();
  }

  async deleteTable(workspaceId, tableId) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables/${tableId}`, {
      method: 'DELETE',
      headers: this.headers
    });

    if (!response.ok) {
      throw new Error(`Failed to delete table: ${response.statusText}`);
    }

    return response.ok;
  }

  async createBIPage(workspaceId, pageData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/biPage`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(pageData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create BI page: ${response.statusText}`);
    }

    return await response.json();
  }

  async addUsersToWorkspace(workspaceId, users) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/bulkUsers`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(users)
    });

    if (!response.ok) {
      throw new Error(`Failed to add users: ${response.statusText}`);
    }

    return await response.json();
  }
}

// Usage
const client = new NoCoDBClient('https://api.example.com', 'your-jwt-token');

// Create workspace
const workspace = await client.createWorkspace({
  name: 'My Workspace',
  description: 'A new workspace'
});

// Create table
const table = await client.createTable(workspace.id, {
  tableName: 'Products',
  description: 'Product catalog'
});

// Add data
const rows = await client.addRows(workspace.id, table.id, [
  { name: 'Product A', price: 100 },
  { name: 'Product B', price: 200 }
]);

// Create BI page
const biPage = await client.createBIPage(workspace.id, {
  name: 'Sales Dashboard',
  description: 'Sales analytics dashboard'
});

// Add users to workspace
const users = await client.addUsersToWorkspace(workspace.id, {
  users: [
    { email: 'user1@example.com', role: 'EDITOR' },
    { email: 'user2@example.com', role: 'VIEWER' }
  ],
  sendInvite: true
});

class NoCoDBV1Client {
  constructor(baseUrl, token) {
    this.baseUrl = baseUrl;
    this.token = token;
    this.headers = {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    };
  }

  async createTransformation(transformationData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v1/transformation`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(transformationData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create transformation: ${response.statusText}`);
    }

    return await response.json();
  }

  async createWidget(widgetData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v1/widgets`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(widgetData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create widget: ${response.statusText}`);
    }

    return await response.json();
  }

  async uploadCSVFile(file, tableName, description) {
    const formData = new FormData();
    formData.append('file', file);
    formData.append('tableName', tableName);
    formData.append('description', description);

    const response = await fetch(`${this.baseUrl}/noCo/api/v1/file/upload/csv`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.token}`
      },
      body: formData
    });

    if (!response.ok) {
      throw new Error(`Failed to upload CSV: ${response.statusText}`);
    }

    return await response.json();
  }

  async createWebhook(webhookData) {
    const response = await fetch(`${this.baseUrl}/noCo/api/v1/tables/webhooks`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(webhookData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create webhook: ${response.statusText}`);
    }

    return await response.json();
  }

  async getFunctions() {
    const response = await fetch(`${this.baseUrl}/noCo/api/v1/functions`, {
      headers: this.headers
    });

    if (!response.ok) {
      throw new Error(`Failed to get functions: ${response.statusText}`);
    }

    return await response.json();
  }
}

NoCoDB v2 Python Client​

import requests
import json
from typing import List, Dict, Any

class NoCoDBClient:
    def __init__(self, base_url: str, token: str):
        self.base_url = base_url
        self.token = token
        self.headers = {
            'Authorization': f'Bearer {token}',
            'Content-Type': 'application/json'
        }
    
    def create_workspace(self, workspace_data: Dict[str, Any]) -> Dict[str, Any]:
        """Create a new workspace"""
        response = requests.post(
            f"{self.base_url}/noCo/api/v2/workspaces",
            headers=self.headers,
            json=workspace_data
        )
        response.raise_for_status()
        return response.json()
    
    def create_table(self, workspace_id: str, table_data: Dict[str, Any]) -> Dict[str, Any]:
        """Create a new table in workspace"""
        response = requests.post(
            f"{self.base_url}/noCo/api/v2/workspaces/{workspace_id}/tables",
            headers=self.headers,
            json=table_data
        )
        response.raise_for_status()
        return response.json()
    
    def add_rows_bulk(self, workspace_id: str, table_id: str, rows: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Add multiple rows to a table"""
        response = requests.post(
            f"{self.base_url}/noCo/api/v2/workspaces/{workspace_id}/tables/{table_id}/rows/bulk",
            headers=self.headers,
            json=rows
        )
        response.raise_for_status()
        return response.json()
    
    def get_tables(self, workspace_id: str, page: int = 0, size: int = 20) -> Dict[str, Any]:
        """Get tables in workspace with pagination"""
        response = requests.get(
            f"{self.base_url}/noCo/api/v2/workspaces/{workspace_id}/tables?page={page}&size={size}",
            headers=self.headers
        )
        response.raise_for_status()
        return response.json()
    
    def create_bi_page(self, workspace_id: str, page_data: Dict[str, Any]) -> Dict[str, Any]:
        """Create a BI page"""
        response = requests.post(
            f"{self.base_url}/noCo/api/v2/workspaces/{workspace_id}/biPage",
            headers=self.headers,
            json=page_data
        )
        response.raise_for_status()
        return response.json()

async function handleAPIError(response) {
  if (!response.ok) {
    const errorData = await response.json();
    
    switch (response.status) {
      case 400:
        throw new Error(`Bad Request: ${errorData.message}`);
      case 401:
        throw new Error('Unauthorized: Please check your authentication');
      case 403:
        throw new Error('Forbidden: Insufficient permissions');
      case 404:
        throw new Error('Not Found: Resource not found');
      case 409:
        throw new Error(`Conflict: ${errorData.message}`);
      case 429:
        throw new Error('Rate Limited: Please try again later');
      default:
        throw new Error(`Server Error: ${errorData.message}`);
    }
  }
}

export class NoCoDBClient {
  private baseUrl: string;
  private token: string;
  private headers: Record<string, string>;

  constructor(baseUrl: string, token: string) {
    this.baseUrl = baseUrl;
    this.token = token;
    this.headers = {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    };
  }

  async createWorkspace(workspaceData: Partial<Workspace>): Promise<Workspace> {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(workspaceData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create workspace: ${response.statusText}`);
    }

    return await response.json();
  }

  async getWorkspaces(page: number = 0, size: number = 20): Promise<APIResponse<Workspace>> {
    const response = await fetch(
      `${this.baseUrl}/noCo/api/v2/workspaces?page=${page}&size=${size}`,
      { headers: this.headers }
    );

    if (!response.ok) {
      throw new Error(`Failed to get workspaces: ${response.statusText}`);
    }

    return await response.json();
  }

  async createTable(workspaceId: string, tableData: Partial<Table>): Promise<Table> {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(tableData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create table: ${response.statusText}`);
    }

    return await response.json();
  }

  async getTables(workspaceId: string, page: number = 0, size: number = 20): Promise<APIResponse<Table>> {
    const response = await fetch(
      `${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables?page=${page}&size=${size}`,
      { headers: this.headers }
    );

    if (!response.ok) {
      throw new Error(`Failed to get tables: ${response.statusText}`);
    }

    return await response.json();
  }

  async addRows(workspaceId: string, tableId: string, rows: Partial<Row>[]): Promise<Row[]> {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables/${tableId}/rows/bulk`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(rows)
    });

    if (!response.ok) {
      throw new Error(`Failed to add rows: ${response.statusText}`);
    }

    return await response.json();
  }

  async createBIPage(workspaceId: string, pageData: Partial<BIPage>): Promise<BIPage> {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/biPage`, {
      method: 'POST',
      headers: this.headers,
      body: JSON.stringify(pageData)
    });

    if (!response.ok) {
      throw new Error(`Failed to create BI page: ${response.statusText}`);
    }

    return await response.json();
  }

  async updateTable(workspaceId: string, tableId: string, tableData: Partial<Table>): Promise<Table> {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables/${tableId}`, {
      method: 'PUT',
      headers: this.headers,
      body: JSON.stringify(tableData)
    });

    if (!response.ok) {
      throw new Error(`Failed to update table: ${response.statusText}`);
    }

    return await response.json();
  }

  async deleteTable(workspaceId: string, tableId: string): Promise<boolean> {
    const response = await fetch(`${this.baseUrl}/noCo/api/v2/workspaces/${workspaceId}/tables/${tableId}`, {
      method: 'DELETE',
      headers: this.headers
    });

    if (!response.ok) {
      throw new Error(`Failed to delete table: ${response.statusText}`);
    }

    return response.ok;
  }

export const DataUpload: React.FC<DataUploadProps> = ({
  baseUrl,
  token,
  workspaceId,
  tableId
}) => {
  const [file, setFile] = useState<File | null>(null);
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState(false);

  const { addRows } = useNoCoDB(baseUrl, token);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const selectedFile = e.target.files?.[0];
    if (selectedFile) {
      setFile(selectedFile);
      setError(null);
      setSuccess(false);
    }
  };

  const handleUpload = async () => {
    if (!file) return;

    try {
      setUploading(true);
      setError(null);
      setSuccess(false);
      setProgress(0);

      // Parse CSV file
      const text = await file.text();
      const rows = parseCSV(text);
      
      // Upload in batches
      const batchSize = 100;
      const totalBatches = Math.ceil(rows.length / batchSize);
      
      for (let i = 0; i < totalBatches; i++) {
        const batch = rows.slice(i * batchSize, (i + 1) * batchSize);
        await addRows(workspaceId, tableId, batch);
        setProgress(((i + 1) / totalBatches) * 100);
      }

      setSuccess(true);
      setFile(null);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Upload failed');
    } finally {
      setUploading(false);
      setProgress(0);
    }
  };

  const parseCSV = (csvText: string): Partial<Row>[] => {
    const lines = csvText.split('\n');
    const headers = lines[0].split(',').map(h => h.trim());
    
    return lines.slice(1).map(line => {
      const values = line.split(',').map(v => v.trim());
      const row: Partial<Row> = {};
      
      headers.forEach((header, index) => {
        row[header] = values[index] || '';
      });
      
      return row;
    }).filter(row => Object.values(row).some(v => v !== ''));
  };

  return (
    <div className="data-upload">
      <h3>Upload Data</h3>
      
      {error && (
        <div className="error-message">
          {error}
        </div>
      )}

      {success && (
        <div className="success-message">
          Data uploaded successfully!
        </div>
      )}

      <div className="upload-section">
        <input
          type="file"
          accept=".csv"
          onChange={handleFileChange}
          disabled={uploading}
        />
        
        {file && (
          <div className="file-info">
            <p>Selected file: {file.name}</p>
            <p>Size: {(file.size / 1024).toFixed(2)} KB</p>
          </div>
        )}

        {uploading && (
          <div className="progress-bar">
            <div 
              className="progress-fill"
              style={{ width: `${progress}%` }}
            />
            <span>{progress.toFixed(1)}%</span>
          </div>
        )}

        <button
          onClick={handleUpload}
          disabled={!file || uploading}
        >
          {uploading ? 'Uploading...' : 'Upload Data'}
        </button>
      </div>
    </div>
  );
};

    <div className="bi-page-list">
      <button
        onClick={() => setShowCreateForm(!showCreateForm)}
        disabled={loading}
      >
        {showCreateForm ? 'Cancel' : 'Create New BI Page'}
      </button>

      {showCreateForm && (
        <form onSubmit={handleSubmit} className="create-form">
          <input
            type="text"
            placeholder="Page Name"
            value={formData.name}
            onChange={(e) => setFormData(prev => ({ ...prev, name: e.target.value }))}
            required
          />
          <textarea
            placeholder="Description"
            value={formData.description}
            onChange={(e) => setFormData(prev => ({ ...prev, description: e.target.value }))}
          />
          <select
            value={formData.layout}
            onChange={(e) => setFormData(prev => ({ ...prev, layout: e.target.value as 'GRID' | 'FLEXIBLE' }))}
          >
            <option value="GRID">Grid Layout</option>
            <option value="FLEXIBLE">Flexible Layout</option>
          </select>
          <button type="submit" disabled={loading}>
            {loading ? 'Creating...' : 'Create BI Page'}
          </button>
        </form>
      )}

      <div className="page-items">
        {pages.map(page => (
          <div
            key={page.id}
            className={`page-item ${selectedPage?.id === page.id ? 'selected' : ''}`}
            onClick={() => onSelectPage(page)}
          >
            <h4>{page.name}</h4>
            <p>{page.description}</p>
            <span className="layout">{page.layout}</span>
          </div>
        ))}
      </div>
    </div>
  );
};
